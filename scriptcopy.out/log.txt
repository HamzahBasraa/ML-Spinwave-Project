SetGridsize(600, 200, 20)
SetCellsize(5e-9, 5e-9, 5e-9)
c1_x := 0.2e-6
c1_y := -0.5e-6
c2_x := -0.1e-6
c2_y := 0.3e-6
c3_x := 0.4e-6
c3_y := -0.2e-6
c4_x := -0.3e-6
c4_y := 0.0
c5_x := 0.5e-6
c5_y := 0.1e-6
c6_x := -0.4e-6
c6_y := -0.1e-6
c7_x := 0.0
c7_y := 0.4e-6
length := 0.05e-6
pillar_height := 50e-9
film_thickness := 50e-9
pillar_z := film_thickness/2 + pillar_height/2
cyl1 := cylinder(length, pillar_height).transl(c1_x, c1_y, pillar_z)
cyl2 := cylinder(length, pillar_height).transl(c2_x, c2_y, pillar_z)
cyl3 := cylinder(length, pillar_height).transl(c3_x, c3_y, pillar_z)
cyl4 := cylinder(length, pillar_height).transl(c4_x, c4_y, pillar_z)
cyl5 := cylinder(length, pillar_height).transl(c5_x, c5_y, pillar_z)
cyl6 := cylinder(length, pillar_height).transl(c6_x, c6_y, pillar_z)
cyl7 := cylinder(length, pillar_height).transl(c7_x, c7_y, pillar_z)
pillar_region := cyl1.add(cyl2).add(cyl3).add(cyl4).add(cyl5).add(cyl6).add(cyl7)
defregion(6, pillar_region)
box := cuboid(1e-6, 1e-6, 50e-9).transl(0, 0, -25e-9)
input := cuboid(1e-6, 0.3e-6, 50e-9).transl(-1e-6, 0, -25e-9)
output1 := cuboid(1e-6, 0.3e-6, 50e-9).transl(1e-6, 350e-9, -25e-9)
output2 := cuboid(1e-6, 0.3e-6, 50e-9).transl(1e-6, -350e-9, -25e-9)
multiplexer := input.add(box).add(output1).add(output2)
full_geom := multiplexer.add(pillar_region)
defregion(1, multiplexer)
input_detector := cuboid(0.2e-6, 0.3e-6, 50e-9).transl(-0.75e-6, 0, -25e-9)
output1_detector := cuboid(0.2e-6, 0.3e-6, 50e-9).transl(0.75e-6, 0.35e-6, -25e-9)
output2_detector := cuboid(0.2e-6, 0.3e-6, 50e-9).transl(0.75e-6, -0.35e-6, -25e-9)
defregion(2, input_detector)
defregion(3, output1_detector)
defregion(4, output2_detector)
setgeom(full_geom)
saveas(geom, "multiplexer")
Msat = 1.4e5
Aex = 3.5e-12
alpha = 2e-4
Msat.SetRegion(6, 800e3)
Aex.SetRegion(6, 13e-12)
Alpha.SetRegion(6, 0.02)
Msat.SetRegion(1, 1.45e5)
Aex.SetRegion(1, 3.5e-12)
Alpha.SetRegion(1, 2e-4)
numberofregions := 120
minD := 2e-4
maxD := 0.5
scale_min := 0.9
scale_max := 1.0
k := pow(maxD/minD, 1.0/(numberofregions-1))
m = uniform(1, .1, 0)
B_ext = vector(0, 0, 0.2)
relax()
save(m)
autosave(m, 200e-12)
tableautosave(10e-12)
f := 1e9
B_ext.SetRegion(2, vector(0, 1e-2*sin(2*pi*f*t), 0))
run(2e-9)

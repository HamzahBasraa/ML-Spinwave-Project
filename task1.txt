// a spin wave should propogate  from one side of a rectangle to another

//this is the grid size 
SetGridsize(600,200 , 1)
SetCellsize(5e-9, 5e-9, 3e-9)
//variables for postions max +/- 0.5e-6, should only change by 100e-9 at a time 
//variables to be changed by a machine learning algorithm


a :=0.3e-6
b :=0.1e-6
c :=-0.4e-6
d :=-0.2e-6
//square cutouts
length := 0.1e-6
box1 := cylinder(length,length).transl(a,c,2e-9)
setgeom(box1)
saveas(geom,"floating_cylinder")

box := rect(1e-6,1e-6)
box2 := rect(length,length).transl(d,b,0)

input := rect(1e-6,0.3e-6).transl(-1e-6, 0,0)

output1 := rect(1e-6,0.3e-6).transl(1e-6,350e-9,0)

output2 := rect(1e-6,0.3e-6).transl(1e-6,-350e-9,0)

multiplexer :=input.add(box).add(output1).add(output2).sub(box2)

setgeom(multiplexer)
saveas(geom, "multiplexer")


defregion(1,multiplexer)


//adding damping 

numberofregions := 120                     // number of shells
minD := 2e-4                 // start damping (inner)
maxD := 0.5                  // end damping (outer)
scale_min := 0.9             // innermost scale (smallest)
scale_max := 1.0             // outermost scale (original size)

// growth factor to go from minD -> maxD in N steps:
k := pow(maxD/minD, 1.0/(numberofregions-1))   // NOTE: maxD/minD, and float divisor

// loop from outermost (largest scale, i = N-1) down to innermost (i = 0)
for i := numberofregions-1; i >= 0; i-- {

    // linear spacing for geometry (physical thickness equal)
    distance := scale_min + (scale_max - scale_min) * i / (numberofregions-1)

    // exponential damping values from minD -> maxD
    alphaValue := minD * pow(k, i)

    regionId := 5 + i

    // scaled geometry
    shape_i := multiplexer.scale(distance, distance, 1)

    defregion(regionId, shape_i)
    alpha.SetRegion(regionId, alphaValue)

}


//creating rectangle shape



//regions for propgation and detection

input_detector := rect(0.2e-6, 0.3e-6).transl(-0.75e-6, 0 , 0)

output1_detector := rect(0.2e-6, 0.3e-6).transl(0.75e-6, 0.35e-6 , 0)

output2_detector := rect(0.2e-6, 0.3e-6).transl(0.75e-6, -0.35e-6 , 0)

defregion(2,input_detector)
defregion(3,output1_detector)
defregion(4,output2_detector)

//material details - YIG
Msat  = 1.4e5
Aex   = 3.5e-12
alpha = 2e-4


// this is how the position of the cells starts
m = uniform(1, .1, 0)
B_ext= vector(0, 0, 0.2)
relax()
save(m)    // relaxed state


//how frequently data should be saved in picoseconds
autosave(m, 200e-12)
tableautosave(10e-12)

//frequency 
f := 1e9 // 1 gigahertz
// the external magnet 
// Uniform bias field (applies everywhere)


// Excitation field (applies only in region 1)
B_ext.SetRegion(2,vector(0, 1e-3*sin(2*pi*f*t), 0))



run(2e-9)

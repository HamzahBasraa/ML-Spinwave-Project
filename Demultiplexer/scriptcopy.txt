length := 0.05e-6

// cylindrical pillars 

pillar_height := 50e-9
film_thickness := 50e-9

pillar_z := film_thickness/2 + pillar_height/2  // = +50 nm

cyl1 := cylinder(length, pillar_height).transl(c1_x, c1_y, pillar_z)
cyl2 := cylinder(length,pillar_height).transl(c2_x,c2_y,pillar_z)
cyl3 := cylinder(length,pillar_height).transl(c3_x,c3_y,pillar_z)
cyl4 := cylinder(length,pillar_height).transl(c4_x,c4_y,pillar_z)

cyl8 := cylinder(length,pillar_height).transl(c8_x,c8_y,pillar_z)
cyl9 := cylinder(length,pillar_height).transl(c9_x,c9_y,pillar_z)

pillar_region := cyl1.add(cyl2).add(cyl3).add(cyl4).add(cyl5).add(cyl6).add(cyl7).add(cyl8).add(cyl9).add(cyl10)
defregion(6,pillar_region)

box := cuboid(1e-6,1e-6,50e-9).transl(0,0,0)

input := cuboid(1e-6,0.3e-6,50e-9).transl(-1e-6, 0,0)

output1 := cuboid(1e-6,0.3e-6,50e-9).transl(1e-6,350e-9,0)

output2 := cuboid(1e-6,0.3e-6,50e-9).transl(1e-6,-350e-9,0)

multiplexer :=input.add(box).add(output1).add(output2)

full_geom := multiplexer.add(pillar_region)

defregion(1,multiplexer)


//adding damping 


//creating rectangle shape



//regions for propgation and detection

input_detector := cuboid(0.2e-6, 0.3e-6,50e-9).transl(-0.75e-6, 0 , 0)

output1_detector := cuboid(0.2e-6, 0.3e-6,50e-9).transl(0.75e-6, 0.35e-6 , 0)

output2_detector := cuboid(0.2e-6, 0.3e-6,50e-9).transl(0.75e-6, -0.35e-6 , 0)





setgeom(full_geom)
saveas(geom, "multiplexer")

//material details - YIG
Msat  = 1.4e5
Aex   = 3.5e-12
alpha = 2e-4

//permalloy material 
Msat.SetRegion(6, 8e5)
Aex.SetRegion(6, 1.3e-11)
alpha.SetRegion(6, 0.013)

Msat.SetRegion(1, 1.45e5)
Aex.SetRegion(1, 3.5e-12)
alpha.SetRegion(1, 2e-4)

numberofregions := 120
minD := 2e-4
maxD := 0.5
scale_min := 0.9
scale_max := 1.0
k := pow(maxD/minD, 1.0/(numberofregions-1))

for i := numberofregions-1; i >= 0; i-- {
    distance := scale_min + (scale_max - scale_min) * i / (numberofregions-1)
    alphaValue := minD * pow(k, i)
    regionId := 12 + i
    
    // Scale the film shape
    shape_i := multiplexer.scale(distance, distance, 1)
    
    // Define the region (Overwrites underlying Region 1)
    defregion(regionId, shape_i)
    
    // CRITICAL FIX: Re-apply Film Msat and set Alpha
    Msat.SetRegion(regionId, 1.45e5) // Restore the 1.45e5 value!
    Aex.SetRegion(regionId, 3.5e-12)
    alpha.SetRegion(regionId, alphaValue)
}


defregion(2,input_detector)
defregion(3,output1_detector)
defregion(4,output2_detector)
// this is how the position of the cells starts
m = uniform(1, .1, 0)

//how frequently data should be saved in picoseconds
autosave(m, 100e-12)
tableautosave(10e-12)

// Excitation field (applies only in region 2)
B_ext.SetRegion(2,vector(0, 1e-2*sin(2*pi*f1*t)+ 1e-2*sin(2*pi*f2*t), 0.02))


run(15e-9)
